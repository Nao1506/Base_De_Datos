Ejemplos de SQL

Crear una tabla

CREATE TABLE alumnos (
    id INT PRIMARY KEY,
    nombre VARCHAR(50),
    edad INT,
    nota DECIMAL(4,2)
);

Agregar información

INSERT INTO alumnos
VALUES (1, 'Ana', 17, 9);

Modificar información

UPDATE alumnos
SET nota = 10
WHERE id = 1;

Eliminar información

DELETE FROM alumnos
WHERE id = 1;
